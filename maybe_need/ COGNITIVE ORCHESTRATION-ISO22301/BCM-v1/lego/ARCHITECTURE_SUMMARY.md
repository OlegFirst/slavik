## 🎮 ДЕМОНСТРАЦИЯ ИСПОЛЬЗОВАНИЯ

```javascript
const { createCognitiveOrchestration } = require('./cognitive-orchestration');

// Создание системы
const system = await createCognitiveOrchestration({
    client: { maxConcurrentRequests: 100 },
    sandbox: { maxConcurrentExperiments: 3 }
});

// Бизнес-логика BCM
await system.executeBusinessLogic(
    'bcm', 'risk-assessment', 'assess',
    { riskId: 'RISK-001' }
);

// Эксперименты с кодом
await system.createExperiment(`
    function optimizedRiskCalculation(risk) {
        return risk.probability * risk.impact * 1.2;
    }
`);

// Аутентификация
await system.authenticateUser({
    username: 'analyst',
    password: 'secure123'
});
```
