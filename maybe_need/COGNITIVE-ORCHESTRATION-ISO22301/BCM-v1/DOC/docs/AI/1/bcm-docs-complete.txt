## 2. Routine Maintenance Schedule

### 2.1 Daily Maintenance Tasks

**System Health Monitoring**
```bash
#!/bin/bash
# daily-health-check.sh
# Run daily at 6:00 AM

echo "=== BCM Platform Daily Health Check ===" | tee -a /var/log/bcm/health.log
date >> /var/log/bcm/health.log

# Check system resources
df -h | grep -E "(bcm|odoo|postgres)" >> /var/log/bcm/health.log
free -h >> /var/log/bcm/health.log

# Check service status
kubectl get pods -n bcm-platform >> /var/log/bcm/health.log

# Check database connections
psql -h $DB_HOST -U $DB_USER -d $DB_NAME -c "SELECT version();" >> /var/log/bcm/health.log

# Check Redis status
redis-cli ping >> /var/log/bcm/health.log

# Send alert if issues found
if [ $? -ne 0 ]; then
    mail -s "BCM Platform Health Check Failed" admin@hospital.com < /var/log/bcm/health.log
fi
```

## 4. Backup Procedures

### 4.1 Database Backups

**Automated Backup Script**
```bash
#!/bin/bash
# database-backup.sh
# Run daily at 1:00 AM

BACKUP_DIR="/backups/database"
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_FILE="bcm_platform_backup_$DATE.sql"

# Create backup directory
mkdir -p $BACKUP_DIR

# Create database dump
pg_dump -h $DB_HOST -U $DB_USER -d $DB_NAME > $BACKUP_DIR/$BACKUP_FILE

# Compress backup
gzip $BACKUP_DIR/$BACKUP_FILE

# Upload to cloud storage (example for AWS S3)
aws s3 cp $BACKUP_DIR/$BACKUP_FILE.gz s3://bcm-backups/database/

# Clean old local backups (keep 7 days)
find $BACKUP_DIR -name "*.sql.gz" -mtime +7 -delete

# Verify backup integrity
if gunzip -t $BACKUP_DIR/$BACKUP_FILE.gz; then
    echo "Backup verification successful: $BACKUP_FILE.gz"
else
    echo "ERROR: Backup verification failed" >&2
    exit 1
fi
```

### 4.2 File System Backups

**Configuration and Data Backup**
```bash
#!/bin/bash
# system-backup.sh

BACKUP_DIR="/backups/system"
DATE=$(date +%Y%m%d_%H%M%S)

# Backup configurations
tar -czf $BACKUP_DIR/config_$DATE.tar.gz /etc/bcm-platform/

# Backup user uploads and attachments
tar -czf $BACKUP_DIR/data_$DATE.tar.gz /var/lib/bcm/

# Backup Kubernetes configurations
kubectl get all -n bcm-platform -o yaml > $BACKUP_DIR/k8s_config_$DATE.yaml

# Upload to cloud storage
aws s3 sync $BACKUP_DIR s3://bcm-backups/system/
```

## 5. Disaster Recovery Testing

### 5.1 Recovery Testing Schedule

**Monthly DR Tests**
```bash
#!/bin/bash
# dr-test.sh
# Test disaster recovery procedures monthly

echo "Starting Disaster Recovery Test - $(date)"

# Test 1: Database Recovery
echo "Testing database recovery..."
TEST_DB="bcm_dr_test_$(date +%Y%m%d)"
LATEST_BACKUP=$(ls -t /backups/database/*.sql.gz | head -1)

createdb -h $DB_HOST -U $DB_USER $TEST_DB
gunzip -c $LATEST_BACKUP | psql -h $DB_HOST -U $DB_USER -d $TEST_DB

if [ $? -eq 0 ]; then
    echo "✅ Database recovery test passed"
else
    echo "❌ Database recovery test failed"
fi

# Test 2: Application Recovery
echo "Testing application recovery..."
kubectl create namespace bcm-dr-test
kubectl apply -f k8s/test-deployment.yaml -n bcm-dr-test

sleep 60

if kubectl get pods -n bcm-dr-test | grep -q Running; then
    echo "✅ Application recovery test passed"
else
    echo "❌ Application recovery test failed"
fi

# Cleanup test resources
dropdb -h $DB_HOST -U $DB_USER $TEST_DB
kubectl delete namespace bcm-dr-test

echo "Disaster Recovery Test completed - $(date)"
```

## 6. Performance Monitoring

### 6.1 System Metrics Collection

**Performance Monitoring Script**
```bash
#!/bin/bash
# performance-monitoring.sh
# Run every 15 minutes

METRICS_FILE="/var/log/bcm/performance_$(date +%Y%m%d).log"

# System metrics
echo "$(date),CPU,$(top -bn1 | grep load | awk '{printf "%.2f\n", $(NF-2)}')" >> $METRICS_FILE
echo "$(date),Memory,$(free | grep Mem | awk '{printf "%.2f\n", $3/$2 * 100.0}')" >> $METRICS_FILE
echo "$(date),Disk,$(df / | tail -1 | awk '{print $5}' | sed 's/%//')" >> $METRICS_FILE

# Database metrics
DB_CONNECTIONS=$(psql -h $DB_HOST -U $DB_USER -d $DB_NAME -t -c "SELECT count(*) FROM pg_stat_activity;")
echo "$(date),DB_Connections,$DB_CONNECTIONS" >> $METRICS_FILE

# Application metrics
HTTP_RESPONSE=$(curl -o /dev/null -s -w "%{http_code}" http://localhost:8069/health)
echo "$(date),HTTP_Response,$HTTP_RESPONSE" >> $METRICS_FILE

# Redis metrics
REDIS_MEMORY=$(redis-cli info memory | grep used_memory_human | cut -d: -f2)
echo "$(date),Redis_Memory,$REDIS_MEMORY" >> $METRICS_FILE
```

### 7.2 Log Analysis

**Error Detection Script**
```bash
#!/bin/bash
# log-analysis.sh

LOG_DIR="/var/log/bcm"
ERROR_REPORT="/tmp/bcm_errors_$(date +%Y%m%d).txt"

echo "BCM Platform Error Report - $(date)" > $ERROR_REPORT
echo "========================================" >> $ERROR_REPORT

# Check for application errors
grep -i "error\|exception\|failed" $LOG_DIR/application.log | tail -20 >> $ERROR_REPORT

# Check for database errors
grep -i "error\|fatal" $LOG_DIR/database.log | tail -10 >> $ERROR_REPORT

# Check for system errors
journalctl -u bcm-platform --since "24 hours ago" --grep="error" >> $ERROR_REPORT

# Send report if errors found
if [ -s $ERROR_REPORT ]; then
    mail -s "BCM Platform Error Report" admin@hospital.com < $ERROR_REPORT
fi

# Clean up
rm $ERROR_REPORT
```

## 8. Maintenance Windows

### 8.1 Scheduled Maintenance

**Maintenance Window Procedures**
```bash
#!/bin/bash
# maintenance-window.sh

MAINTENANCE_START=$(date)
MAINTENANCE_LOG="/var/log/bcm/maintenance_$(date +%Y%m%d_%H%M%S).log"

echo "Maintenance window started: $MAINTENANCE_START" | tee -a $MAINTENANCE_LOG

# 1. Notify users
curl -X POST http://localhost:8069/api/v1/maintenance/start \
    -H "Content-Type: application/json" \
    -d '{"message": "System maintenance in progress", "duration": "2 hours"}'

# 2. Put system in maintenance mode
kubectl annotate deployment bcm-odoo maintenance="true" -n bcm-platform

# 3. Scale down non-essential services
kubectl scale deployment bcm-worker --replicas=0 -n bcm-platform

# 4. Perform maintenance tasks
echo "Starting maintenance tasks..." | tee -a $MAINTENANCE_LOG

# Database maintenance
psql -h $DB_HOST -U $DB_USER -d $DB_NAME -c "VACUUM FULL;" | tee -a $MAINTENANCE_LOG

# Update system packages
apt-get update && apt-get upgrade -y | tee -a $MAINTENANCE_LOG

# Update container images
docker pull ghcr.io/organization/bcm-odoo:latest | tee -a $MAINTENANCE_LOG

# 5. Scale services back up
kubectl scale deployment bcm-worker --replicas=2 -n bcm-platform

# 6. Remove maintenance mode
kubectl annotate deployment bcm-odoo maintenance- -n bcm-platform

# 7. Verify system health
kubectl rollout status deployment/bcm-odoo -n bcm-platform

# 8. Notify completion
MAINTENANCE_END=$(date)
echo "Maintenance window completed: $MAINTENANCE_END" | tee -a $MAINTENANCE_LOG

curl -X POST http://localhost:8069/api/v1/maintenance/end \
    -H "Content-Type: application/json" \
    -d '{"message": "System maintenance completed successfully"}'

# Send maintenance report
mail -s "BCM Platform Maintenance Report" admin@hospital.com < $MAINTENANCE_LOG
```

```bash
#!/bin/bash
# troubleshoot-database.sh

echo "Diagnosing database connection issues..."

# Test database connectivity
pg_isready -h $DB_HOST -p 5432 -U $DB_USER

# Check connection count
psql -h $DB_HOST -U $DB_USER -d $DB_NAME -c "SELECT count(*) FROM pg_stat_activity;"

# Check for long-running queries
psql -h $DB_HOST -U $DB_USER -d $DB_NAME -c "
SELECT pid, now() - pg_stat_activity.query_start AS duration, query 
FROM pg_stat_activity 
WHERE (now() - pg_stat_activity.query_start) > interval '5 minutes';"

# Kill problematic connections if needed
# psql -h $DB_HOST -U $DB_USER -d $DB_NAME -c "SELECT pg_terminate_backend(pid) FROM pg_stat_activity WHERE state = 'idle' AND state_change < now() - interval '1 hour';"
```

### 9.2 Performance Issues

**Memory Usage Investigation**
```bash
#!/bin/bash
# investigate-memory.sh

echo "Investigating memory usage..."

# Check system memory
free -h

# Check container memory usage
kubectl top pods -n bcm-platform

# Check for memory leaks in Odoo
kubectl exec -it deployment/bcm-odoo -n bcm-platform -- \
    python3 -c "
import psutil
import os
process = psutil.Process(os.getpid())
print(f'Memory usage: {process.memory_info().rss / 1024 / 1024:.2f} MB')
print(f'Memory percent: {process.memory_percent():.2f}%')
"

# Check database memory usage
psql -h $DB_HOST -U $DB_USER -d $DB_NAME -c "
SELECT 
    datname,
    pg_size_pretty(pg_database_size(datname)) as size
FROM pg_database 
ORDER BY pg_database_size(datname) DESC;"
```
