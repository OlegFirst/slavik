```python
    def _generate_journey(
        self,
        processes: List[str],
        industry_data: Dict,
        base_duration: float,
        org_size: str
    ) -> List[Dict]:
        """Generate realistic workflow journey"""
        
        stages = [
            'identify_processes',
            'analyze_dependencies',
            'assess_impact',
            'determine_rto',
            'review_results'
        ]
        
        # Stage complexity weights

            'identify_processes': 1.0,
            'analyze_dependencies': 1.5,
            'assess_impact': 1.3,
            'determine_rto': 1.0,
            'review_results': 0.5
        }
        
        journey = []
        current_date = datetime.utcnow() - timedelta(days=random.randint(30, 365))
        
        for stage in stages:
            # Calculate stage duration
            base_hours = (base_duration / len(stages)) * 24
            complexity = complexity_weights[stage]
            duration = base_hours * complexity * random.uniform(0.8, 1.2)
            
            # AI interventions (more in complex stages)
            ai_interventions = random.randint(0, int(complexity * 3))
            
            # Challenges (sometimes)
            challenges = []
            if random.random() > 0.6:
                challenges = random.sample(
                    industry_data['challenges'],
                    random.randint(1, 2)
                )
            
            stage_data = {
                'stage': stage,
                'started_at': current_date.isoformat(),
                'completed_at': (current_date + timedelta(hours=duration)).isoformat(),
                'duration_hours': round(duration, 2),
                'actions': self._generate_stage_actions(stage, len(processes)),
                'challenges': challenges,
                'ai_interventions': ai_interventions
            }
            
            journey.append(stage_data)
            current_date += timedelta(hours=duration)
        
        return journey

        for industry in self.industries.keys():
            for size in ['small', 'medium', 'large']:
                filtered = [
                    c for c in bia_cases
                    if c['organization_context']['industry'] == industry
                    and c['organization_context']['size'] == size
                ]
                
                if filtered:
                    durations = [c['metrics']['total_duration_days'] for c in filtered]
                    ai_usage = [c['metrics']['ai_usage_count'] for c in filtered]
                    
                    key = f"{industry}_{size}"
                    benchmarks[key] = {
                        'avg_duration_days': round(sum(durations) / len(durations), 1),
                        'avg_ai_usage': round(sum(ai_usage) / len(ai_usage), 1),
                        'sample_size': len(filtered)
                    }
        
        return benchmarks

    def save_to_files(self, dataset: Dict, output_dir: str):
        """Save dataset to JSON files"""
        
        output_path = Path(output_dir)
        output_path.mkdir(parents=True, exist_ok=True)
        
        # Save each component
        for key, data in dataset.items():
            if key != 'metadata':
                filename = output_path / f"{key}.json"
                with open(filename, 'w') as f:
                    json.dump(data, f, indent=2)
                print(f"  ✅ Saved {filename}")
        
        # Save metadata
        with open(output_path / "metadata.json", 'w') as f:
            json.dump(dataset['metadata'], f, indent=2)
        
        print(f"\n📦 All data saved to {output_dir}")
```

## 3. END-TO-END INTEGRATION SCRIPT

```python
# scripts/end_to_end_integration.py

"""
End-to-End Integration Test

Demonstrates full platform workflow:
1. Initialize all services
2. Load seed data
3. Run complete BIA workflow
4. Verify AI/ML integration
5. Check data persistence

Usage:
    docker-compose up -d
    python scripts/end_to_end_integration.py
"""

import asyncio
import httpx
from datetime import datetime
import json
from pathlib import Path

class EndToEndIntegration:
    """
    Complete integration test suite
    """
    
    def __init__(self):
        self.base_url = "http://localhost:8000"
        self.ai_url = "http://localhost:8001"
        self.client = httpx.AsyncClient(timeout=30.0)
        
    async def run_full_test(self):
        """Run complete integration test"""
        
        print("🚀 Starting End-to-End Integration Test")
        print("=" * 60)
        
        # Step 1: Health checks
        print("\n1️⃣  HEALTH CHECKS")
        await self.check_all_services()
        
        # Step 2: Load seed data
        print("\n2️⃣  SEED DATA")
        await self.load_seed_data()
        
        # Step 3: Create organization
        print("\n3️⃣  ORGANIZATION SETUP")
        org_id = await self.create_test_organization()
        
        # Step 4: Start BIA workflow
        print("\n4️⃣  BIA WORKFLOW")
        bia_id = await self.start_bia_workflow(org_id)
        
        # Step 5: AI assistance test
        print("\n5️⃣  AI ADVISOR")
        await self.test_ai_advisor(bia_id)
        
        # Step 6: ML prediction test
        print("\n6️⃣  ML PREDICTIONS")
        await self.test_ml_predictions(bia_id)
        
        # Step 7: Complete workflow
        print("\n7️⃣  WORKFLOW COMPLETION")
        await self.complete_workflow(bia_id)
        
        # Step 8: Verify case collection
        print("\n8️⃣  CASE LIBRARY")
        await self.verify_case_collection(bia_id)
        
        # Step 9: Community features
        print("\n9️⃣  COMMUNITY INTELLIGENCE")
        await self.test_community_features()
        
        # Final summary
        print("\n" + "=" * 60)
        print("✅ END-TO-END TEST COMPLETED SUCCESSFULLY!")
        print("=" * 60)
        
        await self.client.aclose()
    
    async def check_all_services(self):
        """Check health of all services"""
        
        services = {
            'Intelligent Core': f"{self.base_url}/health",
            'AI Orchestrator': f"{self.ai_url}/health",
            'MLflow': 'http://localhost:5000/health',
            'Grafana': 'http://localhost:3000/api/health'
        }
        
        for name, url in services.items():
            try:
                response = await self.client.get(url)
                if response.status_code == 200:
                    print(f"  ✅ {name}: Healthy")
                else:
                    print(f"  ⚠️  {name}: Status {response.status_code}")
            except Exception as e:
                print(f"  ❌ {name}: {str(e)}")
    
    async def load_seed_data(self):
        """Load seed data into system"""
        
        # Check if seed data exists
        seed_path = Path("data/seed")
        
        if not seed_path.exists():
            print("  📦 Generating seed data...")
            import subprocess
            subprocess.run([
                "python", "scripts/seed_data_generator.py",
                "--bia-cases", "50",
                "--output", "data/seed"
            ])
        
        # Load via API
        print("  📥 Loading seed data...")
        
        response = await self.client.post(
            f"{self.base_url}/api/v1/admin/seed-data/load",
            json={"source": "data/seed"}
        )
        
        if response.status_code == 200:
            result = response.json()
            print(f"  ✅ Loaded: {result['cases_loaded']} cases")
            print(f"  ✅ Loaded: {result['annotations_loaded']} annotations")
        else:
            print(f"  ❌ Failed: {response.status_code}")
    
    async def create_test_organization(self) -> str:
        """Create test organization"""
        
        org_data = {
            "name": "Test Hospital Integration",
            "industry": "healthcare",
            "size": "medium",
            "employee_count": 500,
            "region": "europe"
        }
        
        response = await self.client.post(
            f"{self.base_url}/api/v1/organizations",
            json=org_data
        )
        
        if response.status_code == 201:
            org = response.json()
            print(f"  ✅ Created organization: {org['id']}")
            return org['id']
        else:
            raise Exception(f"Failed to create org: {response.text}")
    
    async def start_bia_workflow(self, org_id: str) -> str:
        """Start BIA workflow"""
        
        response = await self.client.post(
            f"{self.base_url}/api/v1/workflows/bia/start",
            json={"organization_id": org_id}
        )
        
        if response.status_code == 201:
            workflow = response.json()
            bia_id = workflow['bia_id']
            print(f"  ✅ Started BIA: {bia_id}")
            print(f"  📍 Current stage: {workflow['current_stage']}")
            return bia_id
        else:
            raise Exception(f"Failed to start BIA: {response.text}")
    
    async def test_ai_advisor(self, bia_id: str):
        """Test AI advisor integration"""
        
        # Ask AI for advice
        query = "What processes should I identify for a hospital?"
        
        response = await self.client.post(
            f"{self.ai_url}/api/v1/advisor/ask",
            json={
                "workflow_id": bia_id,
                "query": query,
                "context": {"module": "bia", "stage": "identify_processes"}
            }
        )
        
        if response.status_code == 200:
            advice = response.json()
            print(f"  ✅ AI Response received")
            print(f"  💡 Used {len(advice.get('sources', []))} knowledge sources")
            print(f"  🤖 Confidence: {advice.get('confidence', 0):.2f}")
            
            # Print first 200 chars of advice
            advice_text = advice.get('response', '')
            print(f"\n  AI Advice:\n  {advice_text[:200]}...\n")
        else:
            print(f"  ❌ AI request failed: {response.status_code}")
    
    async def test_ml_predictions(self, bia_id: str):
        """Test ML predictions"""
        
        response = await self.client.post(
            f"{self.base_url}/api/v1/ml/predict/timeline",
            json={
                "workflow_id": bia_id,
                "horizon_months": 6
            }
        )
        
        if response.status_code == 200:
            prediction = response.json()
            print(f"  ✅ Timeline predicted")
            print(f"  📅 Estimated completion: {prediction['estimated_completion_date'][:10]}")
            print(f"  ⏱️  Total days: {prediction['total_estimated_days']}")
            
            # High risk stages
            high_risk = [
                s['stage'] for s in prediction['timeline']
                if s.get('stuck_risk', 0) > 0.7
            ]
            
            if high_risk:
                print(f"  ⚠️  High risk stages: {', '.join(high_risk)}")
        else:
            print(f"  ⚠️  Prediction unavailable (need training data)")
    
    async def complete_workflow(self, bia_id: str):
        """Simulate workflow completion"""
        
        stages = [
            'identify_processes',
            'analyze_dependencies',
            'assess_impact',
            'determine_rto',
            'review_results'
        ]
        
        for i, stage in enumerate(stages, 1):
            print(f"  {i}/5 Completing stage: {stage}")
            
            # Add sample data for stage
            await self._add_stage_data(bia_id, stage)
            
            # Try to advance
            response = await self.client.post(
                f"{self.base_url}/api/v1/workflows/bia/{bia_id}/advance"
            )
            
            if response.status_code == 200:
                result = response.json()
                if result['advanced']:
                    print(f"     ✅ Advanced to: {result['new_stage']}")
                else:
                    print(f"     ⚠️  Blocked: {result.get('reason')}")
            
            # Small delay
            await asyncio.sleep(1)
        
        print("  ✅ Workflow completed!")
    
    async def _add_stage_data(self, bia_id: str, stage: str):
        """Add sample data for stage"""
        
        # Simplified - add minimal required data
        if stage == 'identify_processes':
            await self.client.post(
                f"{self.base_url}/api/v1/workflows/bia/{bia_id}/processes",
                json={
                    "name": "Emergency Department",
                    "tier": "tier_1",
                    "owner": "Dr. Smith"
                }
            )
    
    async def verify_case_collection(self, bia_id: str):
        """Verify case was collected"""
        
        # Wait a bit for async collection
        await asyncio.sleep(2)
        
        response = await self.client.get(
            f"{self.base_url}/api/v1/cases/by-workflow/{bia_id}"
        )
        
        if response.status_code == 200:
            case = response.json()
            print(f"  ✅ Case collected: {case['case_id']}")
            print(f"  📊 Duration: {case['metrics']['total_duration_days']} days")
            print(f"  🤖 AI interventions: {case['metrics']['ai_usage_count']}")
        else:
            print(f"  ⚠️  Case not yet collected (async process)")
    
    async def test_community_features(self):
        """Test community intelligence features"""
        
        # Get benchmarks
        response = await self.client.get(
            f"{self.base_url}/api/v1/community/benchmarks",
            params={"industry": "healthcare", "size": "medium", "module": "bia"}
        )
        
        if response.status_code == 200:
            benchmarks = response.json()
            print(f"  ✅ Benchmarks available")
            print(f"  📈 Avg duration: {benchmarks['duration']['avg_days']} days")
            print(f"  📊 Sample size: {benchmarks['total_cases']} cases")
        else:
            print(f"  ℹ️  Benchmarks not yet available")
        
        # Get community stats
        response = await self.client.get(
            f"{self.base_url}/api/v1/community/stats/community"
        )
        
        if response.status_code == 200:
            stats = response.json()
            print(f"  ✅ Community stats:")
            print(f"     Cases: {stats['total_contributions']}")
            print(f"     Annotations: {stats['community_annotations']}")

async def main():
    """Run integration test"""
    
    integration = EndToEndIntegration()
    
    try:
        await integration.run_full_test()
    except Exception as e:
        print(f"\n❌ Integration test failed: {str(e)}")
        import traceback
        traceback.print_exc()

if __name__ == "__main__":
    asyncio.run(main())
```
