from services.business_logic import StrategyService
from models.domain import StrategyCreate, StrategyType, StrategyPhase

service = StrategyService(repository)

strategy_data = StrategyCreate(
    tenant_id="org123",
    name="DR Strategy",
    strategy_type=StrategyType.BACKUP_RESTORE,
    strategy_phase=StrategyPhase.RESPONSE,
    objective="Restore within RTO",
    scope=["DB", "API"],
    risk_mitigation=["Data loss"]
)

# Создаёт стратегию И публикует событие planning.strategy.created
result = await service.create_strategy(
    strategy_data=strategy_data,
    created_by="user123"
)

print(f"Created: {result.strategy_number}")
# Лог: INFO: Published planning.strategy.created event for strategy STRAT-2025-ABC123
